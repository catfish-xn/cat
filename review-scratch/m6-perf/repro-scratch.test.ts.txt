function envelopeOf(seed: number, opts: { equip?: boolean; minBattles?: number } = {}): { env: SaveEnvelope; equipped: number } {
  const runId = `review-${seed}`;
  const history = new BattleHistory(runId);
  let match = createMatch(seed);
  let equipped = 0;
  const combatEvents = (events: readonly { domain?: string }[]) => events.filter((e): e is LedgerCombatEvent => e.domain === 'combat');
  const resolve = (state: MatchState) => {
    let guard = 0;
    while (state.phase === 'choice' && guard++ < 12) {
      const choice = state.pendingChoice;
      if (choice?.kind === 'augment') {
        const owned = choice.offers.find(id => augmentNeedsHistory(id));
        if (owned) { state = accepted(selectChoice(state, choice.choiceId, choice.generation, owned)); continue; }
        const slot = choice.rerolledSlots?.findIndex(used => !used) ?? -1;
        if (slot >= 0) { state = accepted(rerollAugment(state, choice.choiceId, choice.generation, slot)); continue; }
      }
      state = resolveM5Choices(state);
    }
    return state;
  };
  const min = opts.minBattles ?? 1;
  const stop = getCatalogRoundById('4-2').ordinal;
  for (let guard = 0; guard < 120 && match.round <= stop && (!match.augmentHistory || history.completedRecords.length < min || (opts.equip && !equipped)); guard++) {
    if (match.phase === 'gameOver') break;
    if (match.phase === 'choice') { match = resolve(match); continue; }
    if (match.phase === 'settlement') {
      const before = match; const advanced = nextRound(match, match.round); if (!advanced.ok) break;
      match = advanced.state; history.observe({ before, after: match, events: combatEvents(advanced.events), reason: 'command' }); continue;
    }
    if (match.phase !== 'preparation') break;
    const before = match; const started = startMatchCombat(match); if (!started.ok) break;
    match = started.state; history.observe({ before, after: match, events: combatEvents(started.events), reason: 'command' });
    let ticks = 0;
    while (match.phase === 'combat') {
      if (opts.equip && match.augmentHistory && ticks === 2) {
        for (const item of match.items.filter(i => i.location.kind === 'inventory')) {
          for (const unit of match.preparation.units.filter(u => u.team === 'player' && u.location.kind === 'board')) {
            for (let slot = 0; slot < 3; slot++) {
              const r = equipItem(match, item.id, unit.id, slot);
              if (r.ok) { const b = match; match = r.state; history.observe({ before: b, after: match, events: combatEvents(r.events), reason: 'command' }); equipped++; break; }
            }
            if (equipped) break;
          }
          if (equipped) break;
        }
      }
      const tb = match; const stepped = stepMatch(match); match = stepped.state; ticks++;
      history.observe({ before: tb, after: match, events: combatEvents(stepped.events), reason: 'tick' });
    }
  }
  if (!match.augmentHistory || history.completedRecords.length < 1) throw new Error(`seed ${seed} stopped`);
  return { env: { kind: 'hex-autobattler-save', saveFormatVersion: M8_VERSION.saveFormatVersion, replayFormatVersion: M8_VERSION.replayFormatVersion,
    runId, createdAt: '2026-10-06T00:00:00.000Z', match, battles: history.completedRecords, currentBattle: null }, equipped };
}

async function admit(text: string) { return validateFile(blob(text), validateBattleCollection); }
const counters = () => ({ cold: coldBattleReplayCount(), reuse: transcriptBattleReuseCount() });

  it('tamper: event / hashes / result / endTick on compact and plain files are rejected', async () => {

    const mut = (f: (r: any) => void) => { const e = JSON.parse(plain); f(e.battles[0]); return JSON.stringify(e); };
    const cases: Record<string, string> = {
      eventTick: mut(r => { const ev = r.events.find((x: any) => x.type === 'packetDamage') ?? r.events[0]; if (ev.outcome?.hpDamage !== undefined) ev.outcome.hpDamage += 1; else ev.tick += 1; }),
      eventDropLast: mut(r => { r.events.pop(); }),

      eventHash: mut(r => { r.eventHash = r.eventHash.replace(/.$/, c => c === '0' ? '1' : '0'); }),
      result: mut(r => { r.result = r.result === 'win' ? 'loss' : 'win'; }),

      nextEventSeq: mut(r => { r.nextEventSeq += 1; }),

      contextGold: mut(r => { r.context.gold += 1; }),
      // consistent forge: change an event and recompute nothing -> must fail
